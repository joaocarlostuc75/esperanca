    90→/* ─── Animation helpers ─── */
    91→function FadeInSection({
    92→  children,
    93→  className = "",
    94→  delay = 0,
    95→}: {
    96→  children: React.ReactNode;
    97→  className?: string;
    98→  delay?: number;
    99→}) {
   100→  const ref = useRef<HTMLDivElement>(null);
   101→  const isInView = useInView(ref, { once: true, margin: "-80px" });
   102→
   103→  return (
   104→    <motion.div
   105→      ref={ref}
   106→      initial={{ opacity: 0, y: 40 }}
   107→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   108→      transition={{ duration: 0.7, delay, ease: "easeOut" }}
   109→      className={className}
   110→    >
   111→      {children}
   112→    </motion.div>
   113→  );
   114→}
