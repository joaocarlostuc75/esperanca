   107→/* ─── Animation helpers ─── */
   108→function FadeInSection({
   109→  children,
   110→  className = "",
   111→  delay = 0,
   112→}: {
   113→  children: React.ReactNode;
   114→  className?: string;
   115→  delay?: number;
   116→}) {
   117→  const ref = useRef<HTMLDivElement>(null);
   118→  const isInView = useInView(ref, { once: true, margin: "-80px" });
   119→
   120→  return (
   121→    <motion.div
   122→      ref={ref}
   123→      initial={{ opacity: 0, y: 40 }}
   124→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   125→      transition={{ duration: 0.7, delay, ease: "easeOut" }}
   126→      className={className}
   127→    >
   128→      {children}
   129→    </motion.div>
   130→  );
   131→}
