   108→   108→   108→function FadeInSection({
   109→   109→   109→  children,
   110→   110→   110→  className = "",
   111→   111→   111→  delay = 0,
   112→   112→   112→}: {
   113→   113→   113→  children: React.ReactNode;
   114→   114→   114→  className?: string;
   115→   115→   115→  delay?: number;
   116→   116→   116→}) {
   117→   117→   117→  const ref = useRef<HTMLDivElement>(null);
   118→   118→   118→  const isInView = useInView(ref, { once: true, margin: "-80px" });
   119→   119→   119→
   120→   120→   120→  return (
   121→   121→   121→    <motion.div
   122→   122→   122→      ref={ref}
   123→   123→   123→      initial={{ opacity: 0, y: 40 }}
   124→   124→   124→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   125→   125→   125→      transition={{ duration: 0.7, delay, ease: "easeOut" }}
   126→   126→   126→      className={className}
   127→   127→   127→    >
   128→   128→   128→      {children}
   129→   129→   129→    </motion.div>
   130→   130→   130→  );
   131→   131→   131→}
